#ifndef BLAS_OPS
#define BLAS_OPS

include "Blas/BlasDialect.td"

class Blas_Op<string mnemonic, list<Trait> traits = []> :
    Op<Blas_Dialect, mnemonic, traits>;

//===----------------------------------------------------------------------===//
// blas.matvec -- the worked example. Read this first: it is the template for
// the exercise below.
//===----------------------------------------------------------------------===//

def Blas_MatVecOp : Blas_Op<"matvec"> {
  let summary = "matrix-vector product: out += mat * vec";
  let description = [{
    Computes `out[i] += sum_j mat[i, j] * vec[j]`.

    `mat` is a 2D `f32` memref, `vec` and `out` are 1D `f32` memrefs. This op
    is intentionally left unlowerable by any general-purpose pass: it only
    becomes executable once a dedicated lowering pass (see the `LowerBlasTo*`
    passes in this project) rewrites it into standard dialects.
  }];

  let arguments = (ins
    MemRefRankOf<[F32], [2]>:$mat,
    MemRefRankOf<[F32], [1]>:$vec,
    MemRefRankOf<[F32], [1]>:$out);

  let assemblyFormat =
    "$mat `,` $vec `,` $out attr-dict `:` type($mat) `,` type($vec) `,` type($out)";

  let hasVerifier = 1;
}
